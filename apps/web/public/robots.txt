# Omnimob — https://omnimob.app
#
# Este arquivo precisa existir como ARQUIVO em public/: o app é um SPA e o
# vercel.json reescreve tudo para index.html. A Vercel consulta o sistema de
# arquivos antes de aplicar rewrites, então o que estiver aqui vence a regra —
# mas o que não estiver vira HTML com status 200, que o robô não sabe ler.

User-agent: *
Allow: /

# Painel e telas autenticadas. Não têm valor de busca e o robô só veria a tela
# de login, o que gera páginas duplicadas e sem conteúdo no índice.
Disallow: /login
Disallow: /admin
Disallow: /leads
Disallow: /clientes
Disallow: /usuarios
Disallow: /cargos
Disallow: /configuracoes
Disallow: /tipos-imovel
Disallow: /imoveis

# Link mágico do teste grátis: a URL carrega um token de uso único e curta
# duração. Indexar isso não ajuda ninguém e expõe o parâmetro.
Disallow: /comecar

# As vitrines públicas (/vitrine/...) ficam liberadas de propósito: são o
# conteúdo que cada imobiliária quer encontrável.

# O sitemap NÃO é arquivo: é montado pela API (rota `/public/sitemap.xml`) e
# servido neste endereço por reescrita da Vercel — a lista de vitrines muda a
# cada imobiliária que entra, e um arquivo fixo nunca saberia da mais recente.
#
# Dois cuidados, os dois pela mesma regra (um sitemap só vale para URLs do host
# em que ele está publicado):
#
#   · o endereço aqui é o de OMNIMOB.APP, não o da API. Apontar direto para
#     api.omnimob.app faria o Google descartar tudo lá dentro, porque as URLs
#     listadas são deste domínio e o sitemap estaria noutro.
#   · não recoloque um `sitemap.xml` em `public/`. Vale o mesmo que vale para
#     este arquivo, ao contrário: a Vercel lê o sistema de arquivos ANTES dos
#     rewrites, então o estático venceria a reescrita — em silêncio, e o sitemap
#     voltaria a ter só a home.
Sitemap: https://omnimob.app/sitemap.xml
